# Trigger level checks, run with the console command `scripted_tests none` or the
# -run_tests launch argument. Results land in a text file in the Documents folder.
#
# These answer only what a trigger can answer. Whether an effect actually changed
# anything still needs a probe, because success and fail are both trigger blocks.
#
# Every test fails on the first day after the start date, so a run gives an answer
# immediately rather than after decades of simulation.

last_date = "1836.3.1"

tests = {
	# The same question through a generated geographic region, which probe 08 says works.
	mbga_any_province_in_geo_region = {
		acceptable_fail_rate = 0.0
		success = {
			any_province_in_mbga_wallonia = { always = yes }
		}
		fail = {
			game_date > "1836.1.5"
		}
	}

	# State adjacency, the basis of the border rule.
	mbga_any_neighbouring_state = {
		acceptable_fail_rate = 0.0
		success = {
			c:FRA ?= {
				capital = {
					any_neighbouring_state = { always = yes }
				}
			}
		}
		fail = {
			game_date > "1836.1.5"
		}
	}

	# Are the generated geographic regions complete? Wallonia holds eight provinces.
	mbga_geo_region_province_count = {
		acceptable_fail_rate = 0.0
		success = {
			any_province_in_mbga_wallonia = {
				count = 8
				always = yes
			}
		}
		fail = {
			game_date > "1836.1.5"
		}
	}
}
