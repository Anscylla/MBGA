# The play that carries the war goal, so it can be started against a state directly.
#
# The panel's list of demands against a state is a list of diplomatic plays, not of war goals.
# Without one of these the goal exists but can only be added to a play started for some other
# reason, which means declaring for a state in order to ask for part of it.
#
# Its conditions are the base game's for conquering a state, since it wins the same thing and
# only takes less of it.

mbga_dp_demand_provinces = {
	war_goal = mbga_demand_provinces

	texture = "gfx/interface/icons/diplomatic_action_icons/take_province.dds"

	possible = {
		# Turned off with the war goal it carries, in one file. See
		# common/scripted_triggers/zz_mbga_config.txt.
		mbga_wargoal_enabled = yes

		NOT = { is_country_type = decentralized }
		aggressive_diplomatic_plays_permitted = yes
		NOT = { is_subject_of = scope:target_country }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			NOT = { is_subject_of = root }
		}
		diplomatic_play_has_no_relations_blockers = yes
	}

	# The base game's ceiling for conquering a state. This goal can take the whole state, so it
	# is answered as one.
	ai_acceptance_max = 75

	on_weekly_pulse = {}

	on_war_begins = {}

	on_war_end = {}
}
