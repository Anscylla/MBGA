# PRIVATE, apart from mbga_fill_adjacency_from_table, which a lookup table replaces.
#
# Which candidate provinces the demander can reach from its own border.
#
# The engine will not say whether two provinces touch, so each is made the entire territory
# of a probe country and the question is asked about the states they stand on. Built once
# when a state is opened; every arrangement after that is set arithmetic.
#
# The question answers set against set: with several provinces in each probe it reports
# whether any province of one touches any province of the other. Confirmed both ways, on an
# edge that exists and on one that does not, with the probes verified as holding the ground.
#
# A lookup table may answer instead, at no cost at all, and is used when it proves to
# describe the map being played. It is a separate module: with none loaded everything here
# still works.
#
# Country scope. Reads the candidate containers built by mbga_containers.txt.

mbga_fill_adjacency_from_table = {
}

# Country scope. Writes down that every pair inside this state region has been settled, so
# the sweep never runs on it again in this save.
mbga_mark_region_scanned = {
	set_variable = mbga_adj_known
	if = {
		limit = {
			NOT = {
				is_target_in_variable_list = {
					name = mbga_scanned_regions
					target = var:mbga_active_state_region
				}
			}
		}
		add_to_variable_list = { name = mbga_scanned_regions target = var:mbga_active_state_region }
	}
}

# Country scope. The same, and more: the neighbour lists reach past the state's own edge.
#
# Probing can only ever find neighbours among the candidates, which is enough to say what
# connects to what inside the state but says nothing about the ground beyond it. A table
# names every neighbour, so with one the question of whether a province touches the demander
# becomes a question about who owns those neighbours, and no province need be lent out at all.
mbga_mark_region_mapped = {
	mbga_mark_region_scanned = yes
	set_variable = mbga_adj_reaches_outside
	if = {
		limit = {
			NOT = {
				is_target_in_variable_list = {
					name = mbga_mapped_regions
					target = var:mbga_active_state_region
				}
			}
		}
		add_to_variable_list = { name = mbga_mapped_regions target = var:mbga_active_state_region }
	}
}

# Country scope. Fills the candidates' neighbours from the lookup table, if one is loaded
# and it describes the map actually being played.
#
# The table is checked rather than trusted. A province of this map has land neighbours, so a
# table made for it leaves the containers with some; one made for a different map names
# provinces that are elsewhere or nowhere, and the lists stay empty. Islands are counted with
# the answered ones, the table having said how many of them the state holds, and a majority
# settles it. Failing that, everything found is thrown away and the engine is asked instead.
mbga_apply_adjacency_table = {
	save_scope_as = mbga_builder
	set_variable = { name = mbga_adj_filled value = 0 }
	set_variable = { name = mbga_adj_wanted value = 0 }
	set_variable = { name = mbga_adj_islands value = 0 }

	mbga_fill_adjacency_from_table = yes

	if = {
		limit = { has_variable = mbga_adj_from_table }
		every_in_global_list = {
			variable = mbga_candidates
			scope:mbga_builder = { change_variable = { name = mbga_adj_wanted add = 1 } }
			if = {
				limit = { any_in_list = { variable = mbga_adj count >= 1 } }
				scope:mbga_builder = { change_variable = { name = mbga_adj_filled add = 1 } }
			}
		}

		set_variable = { name = mbga_adj_score value = var:mbga_adj_filled }
		change_variable = { name = mbga_adj_score add = var:mbga_adj_islands }
		change_variable = { name = mbga_adj_score multiply = 2 }
		if = {
			limit = { var:mbga_adj_score >= var:mbga_adj_wanted }
			mbga_mark_region_mapped = yes
			debug_log = "MBGA|table|neighbours came from the lookup table, nothing was probed"
		}
		else = {
			every_in_global_list = {
				variable = mbga_candidates
				clear_variable_list = mbga_adj
			}
			remove_variable = mbga_adj_from_table
			debug_log = "MBGA|table|the lookup table does not describe this map, asking the engine"
		}
	}
}

mbga_lend_province_to = {
	if = {
		limit = { exists = c:$TAG$ }
		c:$TAG$ = { save_scope_as = mbga_taker }
		scope:mbga_lend_province = { mbga_transfer_province_to_taker = yes }
	}
	else = {
		create_country = { tag = $TAG$ province = scope:mbga_lend_province }
	}
}

# Province scope, scope:mbga_prober is the demander. Remembers who to give it back to, then
# hands it to MBH so it can be asked about on its own.
mbga_probe_one_province = {
	save_scope_as = mbga_probe_province
	state.owner = { save_scope_as = mbga_probe_owner }
	scope:mbga_prober = {
		add_to_variable_map = {
			name = mbga_probe_owner_of
			key = scope:mbga_probe_province
			value = scope:mbga_probe_owner
		}
		change_variable = { name = mbga_ops_transfers add = 2 }
	}
	save_scope_as = mbga_lend_province
	mbga_lend_province_to = { TAG = MBH }
}

mbga_return_probed_province = {
	scope:mbga_probe_owner = { save_scope_as = mbga_taker }
	scope:mbga_probe_province = { mbga_transfer_province_to_taker = yes }
}

# Country scope. Fills every candidate's mbga_adj with the candidates it touches, and
# marks mbga_touches_home on those that touch what we already held.
mbga_build_adjacency_graph = {
	save_scope_as = mbga_prober
	set_variable = { name = mbga_pairs_tried value = 0 }
	set_variable = { name = mbga_pairs_found value = 0 }
	set_variable = { name = mbga_holds_right value = 0 }

	set_variable = { name = mbga_ops_transfers value = 0 }
	set_variable = { name = mbga_ops_candidates value = 0 }
	set_variable = { name = mbga_touch_home_count value = 0 }

	# The country question, kept as a measurement only. See where it is asked.
	set_variable = { name = mbga_q_country value = 0 }

	# Who owns what changes between one opening and the next; which province touches which
	# does not. The neighbour lists are never thrown away, so a region gives up its shape once
	# and keeps it for the rest of the save.
	every_in_global_list = {
		variable = mbga_candidates
		limit = { NOT = { has_variable = mbga_own_from_start } }
		remove_variable = mbga_touches_home
		remove_variable = mbga_pair_seen
	}

	# Only when the containers for this region were built a moment ago. Filling from the table
	# on top of neighbours already found would enter them twice.
	if = {
		limit = {
			has_variable = mbga_region_is_new
			NOT = { has_variable = mbga_adj_known }
		}
		mbga_apply_adjacency_table = yes
	}

	# Nothing inside a state can touch our territory unless the state itself does, and that
	# is one question about the state rather than one about every province in it.
	if = {
		limit = {
			var:mbga_active_state ?= {
				any_neighbouring_state = { owner ?= scope:mbga_prober }
			}
		}
		# What touches our own ground.
		#
		# Where the neighbours are known past the state's own edge, that is a question about
		# who owns them and costs nothing. Otherwise each candidate is stood up as a country
		# of its own and asked, and that is two province transfers apiece.
		if = {
			limit = { has_variable = mbga_adj_reaches_outside }
			every_in_global_list = {
				variable = mbga_candidates
				limit = { NOT = { has_variable = mbga_own_from_start } }
				save_scope_as = mbga_outer_container
				scope:mbga_prober = { change_variable = { name = mbga_ops_candidates add = 1 } }
				if = {
					limit = {
						any_in_list = {
							variable = mbga_adj
							state.owner ?= scope:mbga_prober
						}
					}
					set_variable = mbga_touches_home
					scope:mbga_prober = { change_variable = { name = mbga_touch_home_count add = 1 } }
				}
			}
		}
		else = {
			every_in_global_list = {
				variable = mbga_candidates
				limit = { NOT = { has_variable = mbga_own_from_start } }
				save_scope_as = mbga_outer_container
				var:mbga_province = { save_scope_as = mbga_outer_province }
				scope:mbga_outer_province = { state.owner = { save_scope_as = mbga_outer_owner } }
				scope:mbga_prober = {
					add_to_variable_map = {
						name = mbga_probe_owner_of
						key = scope:mbga_outer_province
						value = scope:mbga_outer_owner
					}
				}

				scope:mbga_outer_province = { save_scope_as = mbga_lend_province }
				mbga_lend_province_to = { TAG = MBG }
				scope:mbga_prober = {
					change_variable = { name = mbga_ops_candidates add = 1 }
					change_variable = { name = mbga_ops_transfers add = 2 }
				}

				if = {
					limit = {
						scope:mbga_outer_province.state = {
							any_neighbouring_state = { owner ?= scope:mbga_prober }
						}
					}
					scope:mbga_outer_container = { set_variable = mbga_touches_home }
					scope:mbga_prober = { change_variable = { name = mbga_touch_home_count add = 1 } }
				}

				scope:mbga_outer_owner = { save_scope_as = mbga_taker }
				scope:mbga_outer_province = { mbga_transfer_province_to_taker = yes }
			}
		}

		# What touches what, each unordered pair visited once.
		#
		# The pair sweep stands every candidate up as a country of its own, twice per pair, so it
		# is skipped whenever its answers cannot change anything: the neighbours are known
		# already, nothing touches home so nothing is reachable, or everything does and
		# everything is takeable. A state past the threshold with none of that settled is left
		# to be worked out as it is used: the sweep grows with the square of the province
		# count, while a click costs a pass over what is still out of reach.
		if = {
			limit = {
				NOT = { has_variable = mbga_adj_known }
				var:mbga_ops_candidates > mbga_lazy_threshold
			}
			set_variable = mbga_lazy
			debug_log = "MBGA|graph|too large to work out at once, this state is done as it is used"
		}
		else = { remove_variable = mbga_lazy }

		if = {
			limit = {
				NOT = { has_variable = mbga_lazy }
				NOT = { has_variable = mbga_adj_known }
				var:mbga_touch_home_count > 0
				var:mbga_touch_home_count < var:mbga_ops_candidates
			}
			every_in_global_list = {
				variable = mbga_candidates
				limit = { NOT = { has_variable = mbga_own_from_start } }
				save_scope_as = mbga_outer_container
				var:mbga_province = { save_scope_as = mbga_outer_province }
				scope:mbga_outer_province = { state.owner = { save_scope_as = mbga_outer_owner } }
				scope:mbga_outer_province = { save_scope_as = mbga_lend_province }
				mbga_lend_province_to = { TAG = MBG }
				scope:mbga_prober = { change_variable = { name = mbga_ops_transfers add = 2 } }

				every_in_global_list = {
					variable = mbga_candidates
					limit = {
						NOT = { has_variable = mbga_own_from_start }
						NOT = { has_variable = mbga_pair_seen }
					}
					save_scope_as = mbga_inner_container
					var:mbga_province = { save_scope_as = mbga_inner_province }

					if = {
						limit = {
							NOT = { scope:mbga_inner_province = scope:mbga_outer_province }
							NOT = {
								AND = {
									scope:mbga_outer_container = { has_variable = mbga_touches_home }
									scope:mbga_inner_container = { has_variable = mbga_touches_home }
								}
							}
						}
						scope:mbga_inner_province = { state.owner = { save_scope_as = mbga_inner_owner } }
						scope:mbga_prober = {
							add_to_variable_map = {
								name = mbga_probe_owner_of
								key = scope:mbga_inner_province
								value = scope:mbga_inner_owner
							}
						}
						scope:mbga_inner_province = { save_scope_as = mbga_lend_province }
						mbga_lend_province_to = { TAG = MBH }

						scope:mbga_prober = {
							change_variable = { name = mbga_pairs_tried add = 1 }
							change_variable = { name = mbga_ops_transfers add = 2 }
						}
						if = {
							limit = {
								scope:mbga_outer_province = { owner ?= c:MBG }
								scope:mbga_inner_province = { owner ?= c:MBH }
							}
							scope:mbga_prober = { change_variable = { name = mbga_holds_right add = 1 } }
						}

						if = {
							limit = { c:MBG ?= { is_adjacent_to_country = c:MBH } }
							scope:mbga_prober = { change_variable = { name = mbga_q_country add = 1 } }
						}

						if = {
							limit = {
								scope:mbga_outer_province.state ?= {
									any_neighbouring_state = { owner ?= c:MBH }
								}
							}
							scope:mbga_prober = { change_variable = { name = mbga_pairs_found add = 1 } }
							scope:mbga_outer_container = {
								add_to_variable_list = { name = mbga_adj target = scope:mbga_inner_province }
							}
							scope:mbga_inner_container = {
								add_to_variable_list = { name = mbga_adj target = scope:mbga_outer_province }
							}
						}

						scope:mbga_inner_owner = { save_scope_as = mbga_taker }
						scope:mbga_inner_province = { mbga_transfer_province_to_taker = yes }
					}
				}

				scope:mbga_outer_owner = { save_scope_as = mbga_taker }
				scope:mbga_outer_province = { mbga_transfer_province_to_taker = yes }
				scope:mbga_outer_container = { set_variable = mbga_pair_seen }
			}
		}
		else = {
			debug_log = "MBGA|graph|no pair sweep was needed for this state"
		}

		every_in_global_list = {
			variable = mbga_candidates
			remove_variable = mbga_pair_seen
		}
	}
	else = {
		debug_log = "MBGA|graph|this state does not border us, so nothing here was probed"
	}


	# A complete sweep settles the region for good.
	if = {
		limit = { NOT = { has_variable = mbga_lazy } }
		mbga_mark_region_scanned = yes
	}

	mbga_repair_probe_country = yes

	# Same here: nothing was counted when the game is only working out a tooltip.
	if = {
		limit = { has_variable = mbga_pairs_tried }
		if = {
			limit = { var:mbga_pairs_tried = 0 }
			debug_log = "MBGA|graph|no pairs were tried at all"
		}
		else_if = {
			limit = { var:mbga_holds_right = 0 }
			debug_log = "MBGA|graph|the probe countries never held the pair being asked about"
		}
		else_if = {
			limit = { var:mbga_holds_right < var:mbga_pairs_tried }
			debug_log = "MBGA|graph|the probe countries held the right ground only some of the time"
		}
		else_if = {
			limit = { var:mbga_pairs_found = 0 }
			debug_log = "MBGA|graph|both countries held the right ground, yet no pair came back adjacent"
		}
		else = {
			debug_log = "MBGA|graph|adjacency found between candidates"
		}
	}
}

# Country scope. Works out what may be taken from the graph alone. No countries are
# raised, nothing changes hands, so this is free and runs on every click.
mbga_recompute_takeable = {
	save_scope_as = mbga_prober

	every_in_global_list = {
		variable = mbga_candidates
		limit = { NOT = { has_variable = mbga_own_from_start } }
		remove_variable = mbga_takeable
		save_scope_as = mbga_reach_container

		if = {
			limit = { scope:mbga_prober = { has_variable = mbga_unfiltered } }
			set_variable = mbga_takeable
		}
		else_if = {
			limit = { has_variable = mbga_touches_home }
			set_variable = mbga_takeable
		}
		else = {
			every_in_list = {
				variable = mbga_adj
				save_scope_as = mbga_adj_province
				if = {
					limit = {
						scope:mbga_prober = {
							is_target_in_variable_list = {
								name = mbga_taken
								target = scope:mbga_adj_province
							}
						}
					}
					scope:mbga_reach_container = { set_variable = mbga_takeable }
				}
			}
		}
	}

	set_variable = { name = mbga_takeable_count value = 0 }
	every_in_global_list = {
		variable = mbga_candidates
		limit = { has_variable = mbga_takeable }
		scope:mbga_prober = { change_variable = { name = mbga_takeable_count add = 1 } }
	}
}

mbga_enforce_connected_takes = {
	save_scope_as = mbga_prober
	set_variable = { name = mbga_orphans_released value = 0 }

	# Nothing is joined to anything in an unfiltered list, so nothing can be cut off from
	# it either.
	if = {
		limit = { NOT = { has_variable = mbga_unfiltered } }

		every_in_global_list = {
			variable = mbga_candidates
			remove_variable = mbga_supported
		}

		# the provinces that stand on our own border need nothing else to hold them
		every_in_global_list = {
			variable = mbga_candidates
			limit = {
				has_variable = mbga_taken
				has_variable = mbga_touches_home
			}
			set_variable = mbga_supported
		}

		# and support spreads from those, one ring at a time, until it stops spreading
		set_variable = { name = mbga_support_added value = 1 }
		while = {
			limit = { var:mbga_support_added > 0 }
			set_variable = { name = mbga_support_added value = 0 }

			every_in_global_list = {
				variable = mbga_candidates
				limit = {
					has_variable = mbga_taken
					NOT = { has_variable = mbga_supported }
				}
				save_scope_as = mbga_support_container

				# mbga_adj holds provinces, so the supported containers are walked and each is
				# asked whether the province it wraps is in this one's adjacency list.
				every_in_global_list = {
					variable = mbga_candidates
					limit = { has_variable = mbga_supported }
					var:mbga_province = { save_scope_as = mbga_support_neighbour }
					if = {
						limit = {
							scope:mbga_support_container = {
								NOT = { has_variable = mbga_supported }
								is_target_in_variable_list = {
									name = mbga_adj
									target = scope:mbga_support_neighbour
								}
							}
						}
						scope:mbga_support_container = { set_variable = mbga_supported }
						scope:mbga_prober = { change_variable = { name = mbga_support_added add = 1 } }
					}
				}
			}
		}

		# whatever the flood never reached has no way home
		every_in_global_list = {
			variable = mbga_candidates
			limit = {
				has_variable = mbga_taken
				NOT = { has_variable = mbga_supported }
			}
			save_scope_as = mbga_orphan_container
			var:mbga_province = { save_scope_as = mbga_orphan_province }
			var:mbga_owner = { save_scope_as = mbga_taker }
			scope:mbga_orphan_province = { mbga_transfer_province_to_taker = yes }
			scope:mbga_orphan_container = { remove_variable = mbga_taken }
			scope:mbga_prober = {
				remove_list_variable = { name = mbga_taken target = scope:mbga_orphan_province }
				change_variable = { name = mbga_taken_count subtract = 1 }
				change_variable = { name = mbga_orphans_released add = 1 }
			}
		}

		if = {
			limit = { var:mbga_orphans_released > 0 }
			debug_log = "MBGA|graph|provinces were cut off from home and handed back"
		}
	}
}

# Country scope. Only used while a state is worked out as it is used.
#
# A province just taken is our own ground, so whatever touches it has come within reach.
# Each province still out of reach is asked once; one pass is enough, because a province
# that merely came within reach is still somebody else's and opens nothing further.
mbga_extend_after_take = {
	save_scope_as = mbga_prober

	# Ask about the province just taken rather than about our territory as a whole.
	#
	# "Does X touch our land" is true only of the borders we happen to hold, so the answer
	# expires the moment anything changes hands and the same question has to be asked again.
	# "Does X touch P" is a fact about the map: it is recorded on both containers and never
	# asked twice. A pair already settled is skipped, so every take leaves less to do than
	# the last and a state worked through completely ends up costing nothing to click.
	if = {
		limit = { has_variable = mbga_last_taken }
		var:mbga_last_taken = { save_scope_as = mbga_taken_province }

		every_in_global_list = {
			variable = mbga_candidates
			limit = { var:mbga_province ?= scope:mbga_taken_province }
			save_scope_as = mbga_taken_container
		}

		scope:mbga_taken_province = { save_scope_as = mbga_lend_province }
		mbga_lend_province_to = { TAG = MBG }
		change_variable = { name = mbga_ops_transfers add = 2 }

		every_in_global_list = {
			variable = mbga_candidates
			limit = {
				NOT = { has_variable = mbga_own_from_start }
				NOT = { has_variable = mbga_taken }

				# Already within reach, so nothing this could answer would change anything.
				# The neighbours of a province are found when that province is taken, so
				# skipping it now costs nothing later either.
				NOT = { has_variable = mbga_takeable }

				NOT = {
					is_target_in_variable_list = {
						name = mbga_tested
						target = scope:mbga_taken_province
					}
				}
			}
			save_scope_as = mbga_reach_candidate
			add_to_variable_list = { name = mbga_tested target = scope:mbga_taken_province }

			var:mbga_province = { mbga_probe_one_province = yes }
			if = {
				limit = {
					scope:mbga_probe_province.state = {
						any_neighbouring_state = { owner ?= c:MBG }
					}
				}
				scope:mbga_reach_candidate = {
					add_to_variable_list = { name = mbga_adj target = scope:mbga_taken_province }
				}
				scope:mbga_taken_container ?= {
					add_to_variable_list = { name = mbga_adj target = scope:mbga_probe_province }
				}
			}
			mbga_return_probed_province = yes
		}

		scope:mbga_prober = { save_scope_as = mbga_taker }
		scope:mbga_taken_province = { mbga_transfer_province_to_taker = yes }
	}

	mbga_repair_probe_country = yes
	mbga_recompute_takeable = yes
}

# Country scope. Only used while a state is worked out as it is used.
#
# Without a graph, whether a taken province still hangs on to our ground is asked of the
# engine: it is lent away for a moment and asked whether it still touches us. The pass
# repeats until it releases nothing, because releasing one can strand the next.
mbga_release_orphans_lazily = {
	save_scope_as = mbga_prober
	set_variable = { name = mbga_orphans_released value = 0 }
	set_variable = { name = mbga_lazy_pass value = 1 }

	while = {
		limit = { var:mbga_lazy_pass > 0 }
		set_variable = { name = mbga_lazy_pass value = 0 }

		every_in_global_list = {
			variable = mbga_candidates
			limit = {
				has_variable = mbga_taken
				# A province standing on our own border cannot be cut off from it, so it is
				# never worth asking about.
				NOT = { has_variable = mbga_touches_home }
			}
			save_scope_as = mbga_orphan_container

			var:mbga_province = { mbga_probe_one_province = yes }

			if = {
				limit = {
					NOT = {
						scope:mbga_probe_province.state = {
							any_neighbouring_state = { owner ?= scope:mbga_prober }
						}
					}
				}
				scope:mbga_orphan_container = {
					var:mbga_owner = { save_scope_as = mbga_taker }
					remove_variable = mbga_taken
				}
				scope:mbga_probe_province = { mbga_transfer_province_to_taker = yes }
				scope:mbga_prober = {
					remove_list_variable = { name = mbga_taken target = scope:mbga_probe_province }
					change_variable = { name = mbga_taken_count subtract = 1 }
					change_variable = { name = mbga_orphans_released add = 1 }
					change_variable = { name = mbga_lazy_pass add = 1 }
				}
			}
			else = {
				scope:mbga_prober = { save_scope_as = mbga_taker }
				scope:mbga_probe_province = { mbga_transfer_province_to_taker = yes }
			}
		}
	}

	mbga_repair_probe_country = yes

	if = {
		limit = { var:mbga_orphans_released > 0 }
		debug_log = "MBGA|graph|provinces were cut off from home and handed back"
	}
}

# Container scope, scope:mbga_prober is the demander. Asks about this one province and
# nothing else.
#
# While a state is worked out as it is used, a province that is not yet known to be within
# reach is checked when somebody tries to take it. One question, two transfers, instead of
# asking every province in the state each time anything changes hands. What may be taken is
# exactly the same either way; only the moment of asking differs.
mbga_verify_one_candidate = {
	save_scope_as = mbga_verify_container

	var:mbga_province = { mbga_probe_one_province = yes }
	if = {
		limit = {
			scope:mbga_probe_province.state = {
				any_neighbouring_state = { owner ?= scope:mbga_prober }
			}
		}
		scope:mbga_verify_container = {
			set_variable = mbga_touches_home
			set_variable = mbga_takeable
		}
	}
	mbga_return_probed_province = yes
	mbga_repair_probe_country = yes
}
