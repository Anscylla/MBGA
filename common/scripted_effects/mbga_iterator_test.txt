# Bench for one question: how to reach the provinces of a state without naming a region.
#
# Wired into nothing, run from the decision in common/decisions, and to be deleted with it
# before release. Each candidate is its own effect because an effect the engine refuses is
# discarded whole, taking every branch inside it.

mbga_test_iter_world = {
	every_province_in_old_world = {
		limit = { state ?= scope:mbga_test_state }
		scope:mbga_test_runner = { change_variable = { name = mbga_test_n_world add = 1 } }
	}
	every_province_in_new_world = {
		limit = { state ?= scope:mbga_test_state }
		scope:mbga_test_runner = { change_variable = { name = mbga_test_n_world add = 1 } }
	}
}

# Whether a state region can carry a list. If it can, the map is walked once at the start of
# a game and every province filed under its own region, after which no iterator is needed.
mbga_test_region_store = {
	state_region ?= {
		clear_variable_list = mbga_test_stored
		save_scope_as = mbga_test_region
	}
	every_province_in_old_world = {
		limit = { state ?= scope:mbga_test_state }
		save_scope_as = mbga_test_province
		scope:mbga_test_region = {
			add_to_variable_list = { name = mbga_test_stored target = scope:mbga_test_province }
		}
	}
	every_province_in_new_world = {
		limit = { state ?= scope:mbga_test_state }
		save_scope_as = mbga_test_province
		scope:mbga_test_region = {
			add_to_variable_list = { name = mbga_test_stored target = scope:mbga_test_province }
		}
	}
	scope:mbga_test_region = {
		every_in_list = {
			variable = mbga_test_stored
			scope:mbga_test_runner = { change_variable = { name = mbga_test_n_stored add = 1 } }
		}
	}
}

# Does the oracle answer set against set, or only province against province?
#
# Everything cheaper than the current n squared sweep depends on this. Two provinces are put
# in MBG and two in MBH, and the group answer is compared with the four single answers ORed
# together. Equal on a state with edges and on one without means the oracle is a set test.
#
# Country scope. Uses the first four candidates of the state under test.
mbga_test_pick_four = {
	scope:mbga_test_runner = {
		remove_variable = mbga_test_p1
		remove_variable = mbga_test_p2
		remove_variable = mbga_test_p3
		remove_variable = mbga_test_p4

		# The first two of the list and the last two, on purpose. Candidates are ordered
		# hubs first, so the front of the list sits together and would answer yes whatever
		# the oracle means. Two from each end of a large state are unlikely to touch, and an
		# agreement on no edge is the answer that actually settles the question.
		every_in_list = {
			variable = mbga_test_list
			save_scope_as = mbga_test_pick
			scope:mbga_test_runner = {
				if = {
					limit = { NOT = { has_variable = mbga_test_p1 } }
					set_variable = { name = mbga_test_p1 value = scope:mbga_test_pick }
				}
				else_if = {
					limit = { NOT = { has_variable = mbga_test_p2 } }
					set_variable = { name = mbga_test_p2 value = scope:mbga_test_pick }
				}
				else = {
					if = {
						limit = { has_variable = mbga_test_p4 }
						set_variable = { name = mbga_test_p3 value = var:mbga_test_p4 }
					}
					set_variable = { name = mbga_test_p4 value = scope:mbga_test_pick }
				}
			}
		}

		if = {
			limit = { has_variable = mbga_test_p3 }
			debug_log = "MBGA|test|four provinces picked, two from each end of the state"
		}
		else = {
			debug_log = "MBGA|test|fewer than four provinces to work with, the set question was not asked"
		}
	}
}

# One province to a probe, remembering who to give it back to.
mbga_test_load = {
	var:$SLOT$ = {
		save_scope_as = mbga_lend_province
		state.owner = { save_scope_as = $SLOT$_owner }
	}
	mbga_lend_province_to = { TAG = $TAG$ }
	scope:mbga_test_runner = { change_variable = { name = mbga_test_setcost add = 2 } }
}

mbga_test_unload = {
	scope:$SLOT$_owner = { save_scope_as = mbga_taker }
	var:$SLOT$ = { mbga_transfer_province_to_taker = yes }
}

mbga_test_set_oracle = {
	set_variable = { name = mbga_test_setgroup value = 0 }
	set_variable = { name = mbga_test_setsingles value = 0 }
	set_variable = { name = mbga_test_setcost value = 0 }

	if = {
		limit = { has_variable = mbga_test_p3 }

		# both probes holding two provinces each, asked once
		mbga_test_load = { SLOT = mbga_test_p1 TAG = MBG }
		mbga_test_load = { SLOT = mbga_test_p2 TAG = MBG }
		mbga_test_load = { SLOT = mbga_test_p3 TAG = MBH }
		mbga_test_load = { SLOT = mbga_test_p4 TAG = MBH }
		# An agreement between two answers means nothing if neither probe was holding
		# anything, so the ground is checked before the question is believed.
		set_variable = { name = mbga_test_holds value = 0 }
		if = {
			limit = {
				var:mbga_test_p1 = { state.owner ?= c:MBG }
				var:mbga_test_p2 = { state.owner ?= c:MBG }
				var:mbga_test_p3 = { state.owner ?= c:MBH }
				var:mbga_test_p4 = { state.owner ?= c:MBH }
			}
			set_variable = { name = mbga_test_holds value = 1 }
		}
		if = {
			limit = {
				var:mbga_test_p1 = { state = { any_neighbouring_state = { owner ?= c:MBH } } }
			}
			set_variable = { name = mbga_test_setgroup value = 1 }
		}
		mbga_test_unload = { SLOT = mbga_test_p1 }
		mbga_test_unload = { SLOT = mbga_test_p2 }
		mbga_test_unload = { SLOT = mbga_test_p3 }
		mbga_test_unload = { SLOT = mbga_test_p4 }

		# the same four pairs asked one at a time
		mbga_test_single = { A = mbga_test_p1 B = mbga_test_p3 }
		mbga_test_single = { A = mbga_test_p1 B = mbga_test_p4 }
		mbga_test_single = { A = mbga_test_p2 B = mbga_test_p3 }
		mbga_test_single = { A = mbga_test_p2 B = mbga_test_p4 }

		if = {
			limit = { var:mbga_test_holds = 0 }
			debug_log = "MBGA|test|the probes never held all four, the set question proves nothing"
		}
		else_if = {
			limit = { NOT = { var:mbga_test_setgroup = var:mbga_test_setsingles } }
			debug_log = "MBGA|test|group answer differs from the single answers, not a set test"
		}
		else_if = {
			limit = { var:mbga_test_setgroup = 1 }
			debug_log = "MBGA|test|SET AGAINST SET agreed on an edge that exists"
		}
		else = {
			debug_log = "MBGA|test|SET AGAINST SET agreed that there is no edge, which is the telling case"
		}
	}
}

mbga_test_single = {
	mbga_test_load = { SLOT = $A$ TAG = MBG }
	mbga_test_load = { SLOT = $B$ TAG = MBH }
	if = {
		limit = { var:$A$ = { state = { any_neighbouring_state = { owner ?= c:MBH } } } }
		set_variable = { name = mbga_test_setsingles value = 1 }
	}
	mbga_test_unload = { SLOT = $A$ }
	mbga_test_unload = { SLOT = $B$ }
}

# Country scope. Runs every candidate against the capital and leaves the counts behind.
mbga_test_run_iterators = {
	save_scope_as = mbga_test_runner
	save_scope_as = actor

	set_variable = { name = mbga_test_n_every value = 0 }
	set_variable = { name = mbga_test_n_ordered value = 0 }
	set_variable = { name = mbga_test_n_random value = 0 }
	set_variable = { name = mbga_test_n_regions value = 0 }
	set_variable = { name = mbga_test_n_world value = 0 }
	set_variable = { name = mbga_test_n_stored value = 0 }
	set_variable = { name = mbga_test_n_scopestates value = 0 }
	set_variable = { name = mbga_test_n_fromprov value = 0 }
	remove_variable = mbga_test_seed
	set_variable = { name = mbga_test_n_link value = 0 }
	set_variable = { name = mbga_test_n_strait value = 0 }
	set_variable = { name = mbga_test_n_building value = 0 }
	set_variable = { name = mbga_test_n_sr value = 0 }
	set_variable = { name = mbga_test_n_stateregions value = 0 }
	set_variable = { name = mbga_test_n_states value = 0 }
	set_variable = { name = mbga_test_n_srcap value = 0 }
	set_variable = { name = mbga_test_n_statecap value = 0 }
	set_variable = { name = mbga_test_n_seed value = 0 }
	clear_variable_list = mbga_test_list
	set_variable = mbga_test_ran

	mbga_test_descent = yes
	mbga_test_sr_capital = yes
	mbga_test_strait_all = yes

	if = {
		limit = { exists = capital }
		capital = {
			save_scope_as = mbga_test_state
			mbga_test_iter_every = yes
			mbga_test_iter_ordered = yes
			mbga_test_iter_random = yes
			mbga_test_iter_regions = yes
			mbga_test_iter_world = yes
			mbga_test_region_store = yes
			mbga_test_probe_scopes = yes
			mbga_test_scope_states = yes
			mbga_test_seed_province = yes
			mbga_test_link_province = yes
			mbga_test_link_capital_province = yes
			mbga_test_link_strait = yes
			mbga_test_state_capital_province = yes
			mbga_test_link_building = yes
		}
		mbga_test_pick_four = yes
		mbga_test_seed_check = yes
		mbga_test_set_oracle = yes
		mbga_repair_probe_country = yes
		mbga_test_from_province = yes
		set_variable = { name = mbga_test_state value = scope:mbga_test_state }

		every_in_list = {
			variable = mbga_test_list
			scope:mbga_test_runner = { change_variable = { name = mbga_test_n_regions add = 1 } }
		}

		debug_log = "MBGA|test|iterators run against the capital, see the panel for counts"
		if = {
			limit = { var:mbga_test_n_every > 0 }
			debug_log = "MBGA|test|every_province RETURNED PROVINCES"
		}
		else = {
			debug_log = "MBGA|test|every_province returned nothing, or was thrown out"
		}
		if = {
			limit = { var:mbga_test_n_ordered > 0 }
			debug_log = "MBGA|test|ordered_province RETURNED PROVINCES"
		}
		else = {
			debug_log = "MBGA|test|ordered_province returned nothing, or was thrown out"
		}
		if = {
			limit = { var:mbga_test_n_random > 0 }
			debug_log = "MBGA|test|random_province RETURNED A PROVINCE"
		}
		else = {
			debug_log = "MBGA|test|random_province returned nothing, or was thrown out"
		}
		if = {
			limit = { var:mbga_test_n_world = var:mbga_test_n_regions }
			debug_log = "MBGA|test|the two world halves AGREE with the thirty six branches"
		}
		else = {
			debug_log = "MBGA|test|the two world halves disagree with the thirty six branches"
		}
		if = {
			limit = { var:mbga_test_n_states > 0 }
			debug_log = "MBGA|test|the descent works: strategic regions, state regions and states are all reachable globally"
		}
		else = {
			debug_log = "MBGA|test|the global iterators gave nothing, the descent itself is broken"
		}
		if = {
			limit = { var:mbga_test_n_srcap > 0 }
			debug_log = "MBGA|test|A STRATEGIC REGION HANDS OVER ITS CAPITAL PROVINCE - a seed per region, from nothing"
		}
		if = {
			limit = { var:mbga_test_n_seed = 0 }
			debug_log = "MBGA|test|no seed was ever taken, so the province side probes proved nothing"
		}
		if = {
			limit = { var:mbga_test_n_link > 0 }
			debug_log = "MBGA|test|A STATE HAS A DIRECT PROVINCE LINK - zero names is possible"
		}
		if = {
			limit = { var:mbga_test_n_building > 0 }
			debug_log = "MBGA|test|A BUILDING KNOWS ITS PROVINCE - zero names is possible"
		}
		if = {
			limit = { var:mbga_test_n_fromprov = var:mbga_test_n_regions }
			debug_log = "MBGA|test|every_province FROM A PROVINCE walks the whole state - zero names is possible"
		}
		else_if = {
			limit = { var:mbga_test_n_fromprov > 0 }
			debug_log = "MBGA|test|every_province from a province answered, but with a different count"
		}
		else = {
			debug_log = "MBGA|test|every_province from a province gave nothing either"
		}
		if = {
			limit = { var:mbga_test_n_stored = var:mbga_test_n_regions }
			debug_log = "MBGA|test|a state region CAN hold the list, the index is possible"
		}
		else = {
			debug_log = "MBGA|test|a state region did not hold the list"
		}
	}
	else = {
		debug_log = "MBGA|test|this country has no capital, nothing was tested"
	}
}
