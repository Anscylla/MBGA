# The work behind mbga_open_for_state, split out so the filter toggle can redo it
# without the interface having to hand the state in again.

mbga_open_for_state_effect = {
		debug_log = "MBGA|dyn|a state was handed in from the panel"
		scope:actor = {
			set_variable = { name = mbga_active_state value = ROOT }
			set_variable = { name = mbga_active_state_region value = ROOT.state_region }
			clear_variable_list = mbga_taken
			set_variable = { name = mbga_taken_count value = 0 }
		}
		mbga_fill_selectable_provinces = yes
		scope:actor = {
			set_variable = { name = mbga_loaded_count value = 0 }
			every_in_list = {
				variable = mbga_selectable
				scope:actor = { change_variable = { name = mbga_loaded_count add = 1 } }
			}
			mbga_build_candidates = yes
		}
		# Say the count out loud. Three probes have now drawn a conclusion from a list
		# that was silently empty, so the real code reports what it actually found.
		if = {
			limit = { scope:actor = { var:mbga_candidate_count > 0 } }
			debug_log = "MBGA|dyn|candidates built for this state"
		}
		else = {
			debug_log = "MBGA|dyn|NO candidates - the selectable list was empty"
		}
}

# A province that changes hands moves the frontier: taking one extends our reach to its
# own neighbours, handing it back withdraws it. So the candidate list and every takeable
# flag on it are rebuilt after each change, keeping what is on offer honest as the border
# moves. The provinces already taken are remembered and marked again.
mbga_refresh_after_change = {
	save_scope_as = actor
	if = {
		limit = { has_variable = mbga_active_state }
		clear_variable_list = mbga_previously_taken
		every_in_list = {
			variable = mbga_taken
			save_scope_as = mbga_keep
			scope:actor = { add_to_variable_list = { name = mbga_previously_taken target = scope:mbga_keep } }
		}

		var:mbga_active_state = { mbga_fill_selectable_provinces = yes }
		mbga_build_candidates = yes

		every_in_global_list = {
			variable = mbga_candidates
			if = {
				limit = {
					scope:actor = {
						is_target_in_variable_list = { name = mbga_previously_taken target = prev.var:mbga_province }
					}
				}
				set_variable = mbga_taken
				set_variable = mbga_takeable
			}
		}
		clear_variable_list = mbga_previously_taken
	}
}
