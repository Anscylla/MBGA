# PUBLIC. The front door: opens a state for demanding. See README.md.
#
# mbga_open_for_state_effect  state scope, scope:actor is the demanding country. Records what
#                             was already ours, builds one container per province into the
#                             global list mbga_candidates, and works out what may be taken.
#
# Called by the debug panel, by the diplomatic action, or by any other interface driving the
# mod. Everything it needs, it takes from its own scope.

mbga_open_for_state_effect = {
	save_scope_as = mbga_opened_state
	debug_log = "MBGA|dyn|a state was handed in"
	scope:actor = {
		set_variable = { name = mbga_active_state value = scope:mbga_opened_state }
		set_variable = { name = mbga_active_state_region value = scope:mbga_opened_state.state_region }
		clear_variable_list = mbga_taken
		set_variable = { name = mbga_taken_count value = 0 }
	}
	mbga_fill_selectable_provinces = yes

	scope:actor = {
		clear_variable_list = mbga_own_at_start
		every_in_list = {
			variable = mbga_selectable
			limit = { owner ?= scope:actor }
			save_scope_as = mbga_already_ours
			scope:actor = {
				add_to_variable_list = { name = mbga_own_at_start target = scope:mbga_already_ours }
			}
		}
	}

	scope:actor = {
		set_variable = { name = mbga_loaded_count value = 0 }
		every_in_list = {
			variable = mbga_selectable
			scope:actor = { change_variable = { name = mbga_loaded_count add = 1 } }
		}
		set_variable = { name = mbga_orphans_released value = 0 }
		mbga_build_candidates = yes

		if = {
			limit = { NOT = { has_variable = mbga_unfiltered } }
			mbga_build_adjacency_graph = yes
		}
		mbga_recompute_takeable = yes
	}
	# Only worth saying when the pass actually ran: the game evaluates this effect to
	# build tooltips as well, and nothing is counted then.
	if = {
		limit = { scope:actor = { has_variable = mbga_candidate_count } }
		if = {
			limit = { scope:actor = { var:mbga_candidate_count > 0 } }
			debug_log = "MBGA|dyn|candidates built for this state"
		}
		else_if = {
			limit = { scope:actor = { var:mbga_loaded_count = 0 } }
			debug_log = "MBGA|dyn|the state listed no provinces at all, so nothing could be built"
		}
		else = {
			debug_log = "MBGA|dyn|provinces were listed but none of them became a candidate"
		}
	}
}

mbga_refresh_after_take = {
	# Nothing is asked here while a state is worked out as it is used: a province proves
	# itself when somebody tries to take it, which costs one question instead of one per
	# province in the state.
	mbga_recompute_takeable = yes
}

# Handing back is the move that can strand something, so the taken set is checked for a
# path home before the panel is told what may be taken next.
mbga_refresh_after_return = {
	if = {
		limit = { has_variable = mbga_lazy }
		mbga_release_orphans_lazily = yes
	}
	else = { mbga_enforce_connected_takes = yes }
	mbga_recompute_takeable = yes
}
