# PROBE 10 - the real dispatcher, end to end: state scope in, scope:actor comes out
# holding that state's provinces. Run against the player's own capital so it works
# whoever is being played.

mbga_probe_10_dispatch = {
	debug_log = "MBGA|10|start"
	save_scope_as = actor
	capital = { mbga_fill_selectable_provinces = yes }

	if = {
		limit = { any_in_list = { variable = mbga_selectable always = yes } }
		debug_log = "MBGA|10|RESULT dispatcher filled the list from the capital state"
	}
	else = {
		debug_log = "MBGA|10|RESULT dispatcher produced an EMPTY list"
	}
}
