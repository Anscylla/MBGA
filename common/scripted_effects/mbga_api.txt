# Every country worth demanding from: it exists, it is not us, and it holds land.
mbga_fill_country_list = {
	save_scope_as = mbga_lister
	clear_variable_list = mbga_countries
	every_country = {
		limit = {
			NOT = { this = scope:mbga_lister }
			any_scope_state = { always = yes }
		}
		save_scope_as = mbga_country_entry
		scope:mbga_lister = {
			add_to_variable_list = { name = mbga_countries target = scope:mbga_country_entry }
		}
	}
}

mbga_list_every_province = {
	set_variable = mbga_unfiltered
}

mbga_list_reachable_only = {
	remove_variable = mbga_unfiltered
}
