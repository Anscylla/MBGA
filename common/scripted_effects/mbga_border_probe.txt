# Cleaning up after the adjacency sweep.
#
# Working out the graph in mbga_adjacency.txt hands provinces to MBG and MBH one at a
# time and takes them straight back. Before a province is lent out its owner is written
# into mbga_probe_owner_of on the demanding country, so a sweep that breaks half way can
# still be undone from that record rather than leaving a probe country holding land.
#
# Never seen to run: the log has said "sweep finished cleanly" every time since the courier
# replaced the baked transfer table. It is kept because the day it is needed is the day
# something has already gone wrong.

# Country scope. Hands back anything MBG is still holding, using the owner recorded
# before the sweep touched it. Runs after every sweep, and can be run on its own if a
# game was left in a bad state.
mbga_repair_probe_country = {
	save_scope_as = mbga_prober

	if = {
		limit = {
			OR = {
				c:MBG ?= { any_scope_state = { always = yes } }
				c:MBH ?= { any_scope_state = { always = yes } }
			}
		}
		every_key_in_variable_map = {
			variable = mbga_probe_owner_of
			save_scope_as = mbga_repair_province
			if = {
				limit = {
					OR = {
						scope:mbga_repair_province = { owner ?= c:MBG }
						scope:mbga_repair_province = { owner ?= c:MBH }
					}
				}
				# Back inside the country before reading its map. Every key handed out by
				# every_key_in_variable_map arrives as the scope, so asking here would ask
				# the province for a map the country holds, and get nothing.
				scope:mbga_prober = {
					"variable_map(mbga_probe_owner_of|scope:mbga_repair_province)" = {
						save_scope_as = mbga_taker
					}
				}
				scope:mbga_repair_province = { mbga_transfer_province_to_taker = yes }
				debug_log = "MBGA|probe|a province was still with MBG and has been returned"
			}
		}
	}

	# Belt and braces: if anything is still with MBG after the repair pass, hand the whole
	# lot to the demander rather than leave a stray country on the map. It cannot act on
	# what it holds, its type has politics, economy, military and events switched off, but
	# a country sitting on land it should not have is still wrong.
	if = {
		limit = { c:MBG ?= { any_scope_state = { always = yes } } }
		c:MBG = { annex = scope:mbga_prober }
		debug_log = "MBGA|probe|MBG was annexed as a last resort, something did not return"
	}
	if = {
		limit = { c:MBH ?= { any_scope_state = { always = yes } } }
		c:MBH = { annex = scope:mbga_prober }
		debug_log = "MBGA|probe|MBH was annexed as a last resort, something did not return"
	}

	# A country object can outlive its last province, so the question is whether MBG holds
	# any ground, not whether the object is still around.
	if = {
		limit = {
			OR = {
				c:MBG ?= { any_scope_state = { always = yes } }
				c:MBH ?= { any_scope_state = { always = yes } }
			}
		}
		debug_log = "MBGA|probe|a probe country STILL holds land that could not be traced back"
	}
	else = {
		debug_log = "MBGA|probe|sweep finished cleanly, neither probe country holds anything"
	}
}
