# One container per candidate province.
#
# A data model item cannot be promoted to a province, which is why this used to run on a
# fixed row of numbered slots. A container can be promoted: CMF renders a variable list of
# containers with datacontext = "[Scope.GetScriptContainer]" and hands one back to a
# scripted gui declaring scope = container. So each province travels to the interface
# wrapped in a container, and the list has no ceiling.
#
# Each container carries:
#   mbga_province  the province itself
#   mbga_owner     who held it when the list was built, so a take can be undone
#   mbga_hub       1 city, 2 port, 3 farm, 4 mine, 5 wood, absent for plain countryside
#   mbga_label     a flag holding the word the interface prints
#   mbga_taken     present once it has been handed over

mbga_clear_candidates = {
	every_container = {
		limit = { has_tag = mbga_candidate }
		destroy_container = yes
	}
	clear_global_variable_list = mbga_candidates
	set_variable = { name = mbga_candidate_count value = 0 }
	set_variable = { name = mbga_taken_count value = 0 }
}

# Country scope. Wraps every province of mbga_selectable, hubs first.
#
# Every province of the state gets a candidate, always. Whether it may actually be taken
# is a flag on the candidate rather than a reason to leave it out, so the interface can
# show what is off limits instead of silently hiding it, and further filters can be added
# later without touching this.
#
# It saves its own scope up front rather than leaning on root: this runs from a scripted
# gui whose root is the state that was clicked, so root here is not the country.
mbga_build_candidates = {
	save_scope_as = mbga_builder
	mbga_clear_candidates = yes
	mbga_load_hubs_of_state = yes
	clear_variable_list = mbga_wrapped

	# hubs first, so the named provinces head the list
	every_in_list = {
		variable = mbga_selectable
		limit = {
			# ?= guards the left side of a comparison, not the right. A state with no
			# port simply has no mbga_hub_port, and comparing against a variable that was
			# never set is an error, so each one is asked for only once it exists.
			OR = {
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_city }
					this ?= scope:mbga_builder.var:mbga_hub_city
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_port }
					this ?= scope:mbga_builder.var:mbga_hub_port
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_farm }
					this ?= scope:mbga_builder.var:mbga_hub_farm
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_mine }
					this ?= scope:mbga_builder.var:mbga_hub_mine
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_wood }
					this ?= scope:mbga_builder.var:mbga_hub_wood
				}
			}
		}
		save_scope_as = mbga_wrap_province
		scope:mbga_builder = { mbga_wrap_province = yes }
	}

	# then the rest
	every_in_list = {
		variable = mbga_selectable
		limit = {
			NOT = {
				scope:mbga_builder = {
					is_target_in_variable_list = { name = mbga_wrapped target = prev }
				}
			}
		}
		save_scope_as = mbga_wrap_province
		scope:mbga_builder = { mbga_wrap_province = yes }
	}

	clear_variable_list = mbga_wrapped
	clear_variable_list = mbga_border_provinces_local

	# Outside godmode the border probe decides which of these may actually be taken.
	if = {
		limit = { NOT = { has_variable = mbga_godmode } }
		mbga_mark_takeable_by_probe = yes
	}
}

# Country scope, scope:mbga_wrap_province is the province being wrapped.
#
# on_created carries nothing in with it: saved scopes and local variables from the
# surrounding chain are not visible inside it, which is why the province and the owner
# silently never landed on the container and the transfer was handed a none scope. So
# on_created only plants a marker, the container is picked back up by that marker, and
# everything else is written from the ordinary effect chain where the scopes do exist.
mbga_wrap_province = {
	scope:mbga_wrap_province = { state.owner = { save_scope_as = mbga_wrap_owner } }

	create_container = {
		tags = { mbga_candidate }
		on_created = {
			set_variable = mbga_fresh
		}
	}

	every_container = {
		limit = {
			has_tag = mbga_candidate
			has_variable = mbga_fresh
		}
		remove_variable = mbga_fresh
		save_scope_as = mbga_new_container
	}

	scope:mbga_new_container = {
		set_variable = { name = mbga_province value = scope:mbga_wrap_province }
		set_variable = { name = mbga_owner value = scope:mbga_wrap_owner }
	}

	# Every candidate is takeable. The border rule has no working test behind it: the only
	# adjacency question a province scope accepts is check_area, which compares surface
	# areas and answers false for any pair, and the interface cannot measure a distance.
	# The flag and its plumbing stay so a rule can be dropped in here later without
	# touching anything around it.
	scope:mbga_new_container = { set_variable = mbga_takeable }

	if = {
		limit = {
			has_variable = mbga_hub_city
			scope:mbga_wrap_province ?= var:mbga_hub_city
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 1 }
			set_variable = { name = mbga_label value = flag:city }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_port
			scope:mbga_wrap_province ?= var:mbga_hub_port
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 2 }
			set_variable = { name = mbga_label value = flag:port }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_farm
			scope:mbga_wrap_province ?= var:mbga_hub_farm
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 3 }
			set_variable = { name = mbga_label value = flag:farmland }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_mine
			scope:mbga_wrap_province ?= var:mbga_hub_mine
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 4 }
			set_variable = { name = mbga_label value = flag:mines }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_wood
			scope:mbga_wrap_province ?= var:mbga_hub_wood
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 5 }
			set_variable = { name = mbga_label value = flag:woodland }
		}
	}
	else = {
		scope:mbga_new_container = {
			set_variable = { name = mbga_label value = flag:countryside }
		}
	}

	add_to_global_variable_list = { name = mbga_candidates target = scope:mbga_new_container }
	add_to_variable_list = { name = mbga_wrapped target = scope:mbga_wrap_province }
	change_variable = { name = mbga_candidate_count add = 1 }
}
