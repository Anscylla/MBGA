# One script container per candidate province.
#
# This is how a province reaches the interface: a data model item cannot be promoted to a
# province, a container can. Each carries mbga_province, mbga_owner, mbga_hub and
# mbga_label, and gains mbga_taken once handed over.
#
# Country scope. Reads mbga_selectable, writes the global list mbga_candidates.

mbga_clear_candidates = {
	every_container = {
		limit = { has_tag = mbga_candidate }
		destroy_container = yes
	}
	clear_global_variable_list = mbga_candidates
	set_variable = { name = mbga_candidate_count value = 0 }
	set_variable = { name = mbga_taken_count value = 0 }
}

# Country scope. Wraps every province of mbga_selectable, hubs first.
#
# Containers are kept between openings of the same state region, because what they carry
# about the map does not change: which province each one is, its hub, its label and the
# neighbours found for it. Only what depends on the current borders is cleared and worked
# out again. Reopening a state that has already been scanned therefore costs nothing.
mbga_reuse_candidates = {
	save_scope_as = mbga_builder
	every_in_global_list = {
		variable = mbga_candidates
		remove_variable = mbga_taken
		remove_variable = mbga_takeable
		remove_variable = mbga_touches_home
		remove_variable = mbga_own_from_start
		save_scope_as = mbga_reused_container

		var:mbga_province = { save_scope_as = mbga_reused_province }
		scope:mbga_reused_province = { state.owner ?= { save_scope_as = mbga_reused_owner } }
		scope:mbga_reused_container = {
			set_variable = { name = mbga_owner value = scope:mbga_reused_owner }
		}

		if = {
			limit = {
				scope:mbga_builder = {
					is_target_in_variable_list = {
						name = mbga_own_at_start
						target = scope:mbga_reused_province
					}
				}
			}
			scope:mbga_reused_container = { set_variable = mbga_own_from_start }
		}
	}
	debug_log = "MBGA|dyn|the candidates for this state were already built, keeping them"
}

mbga_build_candidates = {
	save_scope_as = mbga_builder

	# Same state region as last time and the containers are still standing: keep them, and
	# with them everything already learned about which province touches which.
	if = {
		limit = {
			has_variable = mbga_cached_region
			var:mbga_cached_region ?= var:mbga_active_state_region
			any_in_global_list = { variable = mbga_candidates count >= 1 }
		}
		mbga_reuse_candidates = yes
	}
	else = {
		mbga_build_candidates_fresh = yes
	}
}

mbga_build_candidates_fresh = {
	save_scope_as = mbga_builder
	mbga_clear_candidates = yes
	mbga_load_hubs_of_state = yes
	remove_variable = mbga_adj_known
	set_variable = { name = mbga_cached_region value = var:mbga_active_state_region }
	clear_variable_list = mbga_wrapped

	# hubs first, so the named provinces head the list
	every_in_list = {
		variable = mbga_selectable
		limit = {
			OR = {
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_city }
					this ?= scope:mbga_builder.var:mbga_hub_city
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_port }
					this ?= scope:mbga_builder.var:mbga_hub_port
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_farm }
					this ?= scope:mbga_builder.var:mbga_hub_farm
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_mine }
					this ?= scope:mbga_builder.var:mbga_hub_mine
				}
				AND = {
					scope:mbga_builder = { has_variable = mbga_hub_wood }
					this ?= scope:mbga_builder.var:mbga_hub_wood
				}
			}
		}
		save_scope_as = mbga_wrap_province
		scope:mbga_builder = { mbga_wrap_province = yes }
	}

	every_in_list = {
		variable = mbga_selectable
		save_scope_as = mbga_wrap_province
		if = {
			limit = {
				NOT = {
					scope:mbga_builder = {
						is_target_in_variable_list = {
							name = mbga_wrapped
							target = scope:mbga_wrap_province
						}
					}
				}
			}
			scope:mbga_builder = { mbga_wrap_province = yes }
		}
	}

	clear_variable_list = mbga_wrapped
	clear_variable_list = mbga_border_provinces_local
}

mbga_wrap_province = {
	scope:mbga_wrap_province = { state.owner = { save_scope_as = mbga_wrap_owner } }

	create_container = {
		tags = { mbga_candidate }
		on_created = {
			set_variable = mbga_fresh
		}
	}

	every_container = {
		limit = {
			has_tag = mbga_candidate
			has_variable = mbga_fresh
		}
		remove_variable = mbga_fresh
		save_scope_as = mbga_new_container
	}

	scope:mbga_new_container = {
		set_variable = { name = mbga_province value = scope:mbga_wrap_province }
		set_variable = { name = mbga_owner value = scope:mbga_wrap_owner }
	}

	# Ours before the session started, so there is nothing to demand here. Kept in the
	# list rather than dropped, so the state still shows its full shape.
	if = {
		limit = {
			is_target_in_variable_list = { name = mbga_own_at_start target = scope:mbga_wrap_province }
		}
		scope:mbga_new_container = { set_variable = mbga_own_from_start }
	}

	# Nothing is takeable until something says so. Under the unfiltered rule that is the
	# whole state at once, otherwise it is whatever the border scan reaches, so a province
	# never offers itself by default.
	if = {
		limit = { scope:mbga_builder = { has_variable = mbga_unfiltered } }
		scope:mbga_new_container = { set_variable = mbga_takeable }
	}

	if = {
		limit = {
			has_variable = mbga_hub_city
			scope:mbga_wrap_province ?= var:mbga_hub_city
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 1 }
			set_variable = { name = mbga_label value = flag:city }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_port
			scope:mbga_wrap_province ?= var:mbga_hub_port
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 2 }
			set_variable = { name = mbga_label value = flag:port }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_farm
			scope:mbga_wrap_province ?= var:mbga_hub_farm
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 3 }
			set_variable = { name = mbga_label value = flag:farmland }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_mine
			scope:mbga_wrap_province ?= var:mbga_hub_mine
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 4 }
			set_variable = { name = mbga_label value = flag:mines }
		}
	}
	else_if = {
		limit = {
			has_variable = mbga_hub_wood
			scope:mbga_wrap_province ?= var:mbga_hub_wood
		}
		scope:mbga_new_container = {
			set_variable = { name = mbga_hub value = 5 }
			set_variable = { name = mbga_label value = flag:woodland }
		}
	}
	else = {
		scope:mbga_new_container = {
			set_variable = { name = mbga_label value = flag:countryside }
		}
	}

	add_to_global_variable_list = { name = mbga_candidates target = scope:mbga_new_container }
	add_to_variable_list = { name = mbga_wrapped target = scope:mbga_wrap_province }
	change_variable = { name = mbga_candidate_count add = 1 }
}
