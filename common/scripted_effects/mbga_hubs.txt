# Which province of a state is its city, port, farm, mine or wood hub.
#
# The one thing the engine will not give back. Script can set a hub's name but has no
# trigger or scope link that reads one, and the interface has no Province.GetName to
# compare against State.GetCityHubName either, so the mapping is generated from map_data
# by tools/bake_provinces.py.
#
# It used to be two files: an effect per state region setting five variables, and a
# dispatcher of 781 if/else_if branches choosing which to call. The dispatcher is gone.
# The table is now a variable map keyed by the state region itself, so a state looks its
# own hubs up rather than being compared against every name in the game.
#
# Two things about reading a variable map, both learned the hard way.
#
# The read has to happen in the scope that owns the map. every_key_in_variable_map hands
# out the keys, so inside it the scope is a state region, and asking that for the country's
# map gives "map due to not being set" - the map is fine, it is being asked of the wrong
# thing. So the key is saved and the read is done back inside the country.
#
# And the five kinds are written out one after another rather than passed to one effect as
# a parameter. $KIND$ is not substituted inside a quoted event target link: the game read
# "variable_map(mbga_hubmap_$KIND$|...)" as a map called mbga_hubmapwood$, losing the
# underscore and keeping the dollar, and every lookup failed.
#
# Two things keep this honest on a modded map. Every block in the generated table is
# entered with ?=, the base game's own idiom for "if this still exists", so a renamed or
# deleted state region is skipped rather than raising an error. And every answer is
# checked before it is believed: a hub province whose state region is no longer the one
# that asked for it is discarded, and that state falls back to terrain names. A mod may
# therefore redraw part of the map and keep working hub names everywhere it did not touch.

# Country scope. Fills mbga_hub_city, _port, _farm, _mine and _wood for the state in
# mbga_active_state_region, leaving any of them unset when there is no trustworthy answer.
mbga_load_hubs_of_state = {
	save_scope_as = mbga_hub_reader

	remove_variable = mbga_hub_city
	remove_variable = mbga_hub_port
	remove_variable = mbga_hub_farm
	remove_variable = mbga_hub_mine
	remove_variable = mbga_hub_wood

	# 3072 entries, so it is built once and then answers every state for the rest of the
	# session.
	if = {
		limit = { NOT = { has_variable = mbga_hub_maps_ready } }
		set_variable = mbga_hub_maps_ready
		mbga_fill_hub_maps = yes

		if = {
			limit = { has_variable_map = mbga_hubmap_city }
			debug_log = "MBGA|hubs|the hub table was built"
		}
		else = {
			debug_log = "MBGA|hubs|the hub table came back empty, names fall back to terrain"
		}
	}

	if = {
		limit = { has_variable = mbga_active_state_region }
		var:mbga_active_state_region = { save_scope_as = mbga_hub_region }

		# city
		if = {
			limit = { has_variable_map = mbga_hubmap_city }
			every_key_in_variable_map = {
				variable = mbga_hubmap_city
				limit = { this = scope:mbga_hub_region }
				save_scope_as = mbga_hub_key

				scope:mbga_hub_reader = {
					"variable_map(mbga_hubmap_city|scope:mbga_hub_key)" = {
						save_scope_as = mbga_hub_city_found
					}
					if = {
						limit = {
							scope:mbga_hub_city_found = {
								state.state_region ?= scope:mbga_hub_region
							}
						}
						set_variable = { name = mbga_hub_city value = scope:mbga_hub_city_found }
					}
				}
			}
		}
		# port
		if = {
			limit = { has_variable_map = mbga_hubmap_port }
			every_key_in_variable_map = {
				variable = mbga_hubmap_port
				limit = { this = scope:mbga_hub_region }
				save_scope_as = mbga_hub_key

				scope:mbga_hub_reader = {
					"variable_map(mbga_hubmap_port|scope:mbga_hub_key)" = {
						save_scope_as = mbga_hub_port_found
					}
					if = {
						limit = {
							scope:mbga_hub_port_found = {
								state.state_region ?= scope:mbga_hub_region
							}
						}
						set_variable = { name = mbga_hub_port value = scope:mbga_hub_port_found }
					}
				}
			}
		}
		# farm
		if = {
			limit = { has_variable_map = mbga_hubmap_farm }
			every_key_in_variable_map = {
				variable = mbga_hubmap_farm
				limit = { this = scope:mbga_hub_region }
				save_scope_as = mbga_hub_key

				scope:mbga_hub_reader = {
					"variable_map(mbga_hubmap_farm|scope:mbga_hub_key)" = {
						save_scope_as = mbga_hub_farm_found
					}
					if = {
						limit = {
							scope:mbga_hub_farm_found = {
								state.state_region ?= scope:mbga_hub_region
							}
						}
						set_variable = { name = mbga_hub_farm value = scope:mbga_hub_farm_found }
					}
				}
			}
		}
		# mine
		if = {
			limit = { has_variable_map = mbga_hubmap_mine }
			every_key_in_variable_map = {
				variable = mbga_hubmap_mine
				limit = { this = scope:mbga_hub_region }
				save_scope_as = mbga_hub_key

				scope:mbga_hub_reader = {
					"variable_map(mbga_hubmap_mine|scope:mbga_hub_key)" = {
						save_scope_as = mbga_hub_mine_found
					}
					if = {
						limit = {
							scope:mbga_hub_mine_found = {
								state.state_region ?= scope:mbga_hub_region
							}
						}
						set_variable = { name = mbga_hub_mine value = scope:mbga_hub_mine_found }
					}
				}
			}
		}
		# wood
		if = {
			limit = { has_variable_map = mbga_hubmap_wood }
			every_key_in_variable_map = {
				variable = mbga_hubmap_wood
				limit = { this = scope:mbga_hub_region }
				save_scope_as = mbga_hub_key

				scope:mbga_hub_reader = {
					"variable_map(mbga_hubmap_wood|scope:mbga_hub_key)" = {
						save_scope_as = mbga_hub_wood_found
					}
					if = {
						limit = {
							scope:mbga_hub_wood_found = {
								state.state_region ?= scope:mbga_hub_region
							}
						}
						set_variable = { name = mbga_hub_wood value = scope:mbga_hub_wood_found }
					}
				}
			}
		}
	}
}
