# PROBE 04 - transfer a single province and hand it straight back.
# This is the core mechanic: click a province, it changes hands immediately as a
# preview, and is reverted when the selection is cleared.

mbga_probe_04_transfer = {
	debug_log = "MBGA|04|start"

	p:x9686A5 = {
		save_scope_as = mbga_probe_province
		state = { owner = { save_scope_as = mbga_original_owner } }
	}

	if = {
		limit = { exists = scope:mbga_original_owner }
		debug_log = "MBGA|04|save_scope_as on a province works"

		# hand it to the player
		p:x9686A5.state.state_region = {
			set_owner_of_provinces = {
				country = ROOT
				provinces = { x9686A5 }
			}
		}
		debug_log = "MBGA|04|transferred to player"

		# and hand it straight back
		p:x9686A5.state.state_region = {
			set_owner_of_provinces = {
				country = scope:mbga_original_owner
				provinces = { x9686A5 }
			}
		}
		debug_log = "MBGA|04|reverted to original owner"
	}
	else = {
		debug_log = "MBGA|04|could not resolve original owner"
	}

	debug_log = "MBGA|04|end"
}
