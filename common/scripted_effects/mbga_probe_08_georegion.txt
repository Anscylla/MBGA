# PROBE 08 - the generated geographic region iterator, now running against the real
# baked definitions rather than a hand-written test region.
# STATE_WALLONIA holds 8 provinces.

mbga_probe_08_georegion = {
	debug_log = "MBGA|08|start - expecting 8 provinces in STATE_WALLONIA"
	set_variable = { name = mbga_geo_count value = 0 }

	every_province_in_mbga_wallonia = {
		root = { change_variable = { name = mbga_geo_count add = 1 } }
	}

	if = {
		limit = { var:mbga_geo_count ?= 8 }
		debug_log = "MBGA|08|RESULT baked geographic regions work - counted exactly 8"
	}
	else = {
		debug_log = "MBGA|08|RESULT iterator did not return 8 provinces"
	}

	remove_variable = mbga_geo_count
}
