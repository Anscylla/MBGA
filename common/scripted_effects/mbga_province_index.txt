# PUBLIC. Every province in the game, filed under its state region. See README.md.
#
# mbga_ensure_province_index  any scope. Builds the index if it is missing or out of date.
#                             Cheap once built; every read goes through it.
# mbga_build_province_index   any scope. Builds it unconditionally. Called from on_game_started.
#
# The list mbga_region_provinces on a state region is the index itself, and may be read
# directly. The trigger mbga_province_index_is_ready says whether it is there.
#
# There is no province iterator on a state scope and only a geographic region yields province
# scopes, so the map is walked through old_world and new_world, which between them hold every
# land strategic region. The index lives on state regions rather than states because a state
# is one owner's share of a region and comes and goes with the borders.

mbga_index_one_province = {
	save_scope_as = mbga_index_province
	state.state_region ?= {
		add_to_variable_list = { name = mbga_region_provinces target = scope:mbga_index_province }
		change_global_variable = { name = mbga_index_count add = 1 }
	}
}

# Walks the whole map. Safe to call again: every region is emptied first, so a rebuild
# replaces the index rather than doubling it.
mbga_build_province_index = {
	every_state_region = {
		clear_variable_list = mbga_region_provinces
	}
	set_global_variable = { name = mbga_index_count value = 0 }

	every_province_in_old_world = { mbga_index_one_province = yes }
	every_province_in_new_world = { mbga_index_one_province = yes }

	set_global_variable = mbga_index_built
	set_global_variable = { name = mbga_index_version value = 1 }

	if = {
		limit = { global_var:mbga_index_count > 0 }
		debug_log = "MBGA|index|the province index was built"
	}
	else = {
		debug_log = "MBGA|index|the province index came back EMPTY, nothing will list"
	}
}

mbga_ensure_province_index = {
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = mbga_index_built }
				NOT = { has_global_variable = mbga_index_version }
				global_var:mbga_index_version < 1
				global_var:mbga_index_count = 0
			}
		}
		debug_log = "MBGA|index|no usable index found, building one now"
		mbga_build_province_index = yes
	}
}
