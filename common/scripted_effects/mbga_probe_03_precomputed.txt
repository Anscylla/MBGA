# PROBE 03 - runtime enumeration of a state's provinces is impossible
# ("Inconsistent effect scopes (state vs. province)"), so lists are baked from
# map_data. This proves the baked path works and prints real counts.
# Province ids below are the real STATE_WALLONIA list.

mbga_probe_03_precomputed = {
	debug_log = "MBGA|03|start"

	clear_variable_list = mbga_baked_provinces
	add_to_variable_list = { name = mbga_baked_provinces target = p:x916574 }
	add_to_variable_list = { name = mbga_baked_provinces target = p:xB0A0C0 }
	add_to_variable_list = { name = mbga_baked_provinces target = p:x5180A0 }

	set_variable = { name = mbga_count value = 0 }
	every_in_list = {
		variable = mbga_baked_provinces
		ROOT = { change_variable = { name = mbga_count add = 1 } }
	}
	if = {
		limit = { var:mbga_count ?= 3 }
		debug_log = "MBGA|03|baked list size is exactly 3 as expected"
	}
	else = {
		debug_log = "MBGA|03|baked list size is NOT 3"
	}

	debug_log = "MBGA|03|end"
}
