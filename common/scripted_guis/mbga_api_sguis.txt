# MBGA public interface API. An interface calls these; nothing here reaches into a
# vanilla panel, so any mod can drive the same flow from its own gui.

# Build the list of countries that can be demanded from.
mbga_refresh_countries = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		mbga_fill_country_list = yes
		debug_log = "MBGA|api|country list rebuilt"
	}
}

# Choose who is being demanded from. Root is that country, scope:actor is the demander.
mbga_set_target_country = {
	scope = country
	saved_scopes = { actor }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		scope:actor = {
			set_variable = { name = mbga_target_country value = root }
			remove_variable = mbga_active_state
			clear_variable_list = mbga_selectable
			clear_variable_list = mbga_taken
			set_variable = { name = mbga_taken_count value = 0 }
			set_variable = { name = mbga_loaded_count value = 0 }
			mbga_clear_candidates = yes
		}
		debug_log = "MBGA|api|target country set"
	}
}

# True once a target country has been chosen, so an interface can reveal its state list.
mbga_has_target_country = {
	scope = country
	is_shown = { has_variable = mbga_target_country }
	is_valid = { always = yes }
	effect = {}
}

# Whether this mod's own province tool is on screen, which is one game rule and nothing
# else. The panel hangs off this.
#
# It says nothing about which provinces may be demanded. That is a rule of its own, and it
# reaches the mechanism as the API's own choice of list through mbga_list_every_province and
# mbga_list_reachable_only, made in mbga_state_sgui.txt where this mod's interface meets the
# mechanism. The two are deliberately independent: the tool can be open while demands are
# still restricted, and demands can be unrestricted with no tool in sight.
mbga_tool_enabled = {
	scope = country
	is_shown = { has_game_rule = mbga_province_tool_debug }
	is_valid = { always = yes }
	effect = {}
}

# Show or hide the numbers beside the panel.
mbga_toggle_stats = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = mbga_stats_shown }
			remove_variable = mbga_stats_shown
		}
		else = {
			set_variable = mbga_stats_shown
		}
	}
}

mbga_stats_shown = {
	scope = country
	is_shown = { has_variable = mbga_stats_shown }
	is_valid = { always = yes }
	effect = {}
}

# The iterator bench has been run at least once, so its readout may show.
mbga_test_has_run = {
	scope = country
	is_shown = { has_variable = mbga_test_ran }
	is_valid = { always = yes }
	effect = {}
}

# Done picking. Clears the target and the candidates, which takes the markers off the map.
#
# The debug panel has a whole window to close; the diplomatic route has nothing but markers,
# so it needs somewhere to say it is finished.
mbga_close_tool = {
	scope = country
	is_shown = { has_variable = mbga_target_country }
	is_valid = { always = yes }
	effect = {
		mbga_clear_candidates = yes
		remove_variable = mbga_target_country
		remove_variable = mbga_active_state
		remove_variable = mbga_active_state_region
		clear_variable_list = mbga_selectable
		debug_log = "MBGA|api|the tool was put down"
	}
}

# The diplomatic route is in use, so the little bar that ends it should show.
mbga_tool_is_diplomatic = {
	scope = country
	is_shown = { has_game_rule = mbga_province_tool_diplomatic }
	is_valid = { always = yes }
	effect = {}
}
