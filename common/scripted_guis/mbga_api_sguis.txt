# PUBLIC. What an interface may ask of the mod: the country list, the target country, the
# tool's own state and the numbers panel. See README.md.
#
# mbga_toggle_stats, mbga_stats_shown and mbga_test_has_run belong to the debug panel and are
# not part of the contract.

mbga_refresh_countries = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		mbga_fill_country_list = yes
	}
}

# Choose who is being demanded from. Root is that country, scope:actor is the demander.
mbga_set_target_country = {
	scope = country
	saved_scopes = { actor }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		scope:actor = {
			set_variable = { name = mbga_target_country value = root }
			remove_variable = mbga_active_state
			clear_variable_list = mbga_selectable
			clear_variable_list = mbga_taken
			set_variable = { name = mbga_taken_count value = 0 }
			set_variable = { name = mbga_loaded_count value = 0 }
			mbga_clear_candidates = yes
		}
	}
}

# True once a target country has been chosen, so an interface can reveal its state list.
mbga_has_target_country = {
	scope = country
	is_shown = { has_variable = mbga_target_country }
	is_valid = { always = yes }
	effect = {}
}

mbga_tool_enabled = {
	scope = country
	is_shown = { has_game_rule = mbga_province_tool_debug }
	is_valid = { always = yes }
	effect = {}
}

# Show or hide the numbers beside the panel.
mbga_toggle_stats = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = mbga_stats_shown }
			remove_variable = mbga_stats_shown
		}
		else = {
			set_variable = mbga_stats_shown
		}
	}
}

mbga_stats_shown = {
	scope = country
	is_shown = { has_variable = mbga_stats_shown }
	is_valid = { always = yes }
	effect = {}
}

# Roll the list of provinces out of the bottom of the settlement window, or back in.
mbga_toggle_province_list = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = mbga_list_shown }
			remove_variable = mbga_list_shown
		}
		else = {
			set_variable = mbga_list_shown
		}
	}
}

mbga_province_list_shown = {
	scope = country
	is_shown = { has_variable = mbga_list_shown }
	is_valid = { always = yes }
	effect = {}
}

# The iterator bench has been run at least once, so its readout may show.
mbga_test_has_run = {
	scope = country
	is_shown = { has_variable = mbga_test_ran }
	is_valid = { always = yes }
	effect = {}
}

mbga_close_tool = {
	scope = country
	is_shown = { has_variable = mbga_target_country }
	is_valid = { always = yes }
	effect = {
		mbga_finish_settlement = yes
		mbga_clear_candidates = yes
		remove_variable = mbga_target_country
		remove_variable = mbga_active_state
		remove_variable = mbga_active_state_region
		clear_variable_list = mbga_selectable
		debug_log = "MBGA|api|the tool was put down"
	}
}

# A war is being settled rather than a demand made, so the bar shows what it will cost.
mbga_is_settling = {
	scope = country
	is_shown = { has_variable = mbga_settling }
	is_valid = { always = yes }
	effect = {}
}

# The tool is on at all, whichever way it is reached. The bar that ends a selection belongs
# to the tool rather than to one of its settings.
mbga_tool_is_on = {
	scope = country
	is_shown = {
		OR = {
			has_game_rule = mbga_province_tool_diplomatic
			has_game_rule = mbga_province_tool_debug
		}
	}
	is_valid = { always = yes }
	effect = {}
}
