# MBGA public interface API. An interface calls these; nothing here reaches into a
# vanilla panel, so any mod can drive the same flow from its own gui.

# Build the list of countries that can be demanded from.
mbga_refresh_countries = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		mbga_fill_country_list = yes
		debug_log = "MBGA|api|country list rebuilt"
	}
}

# Choose who is being demanded from. Root is that country, scope:actor is the demander.
mbga_set_target_country = {
	scope = country
	saved_scopes = { actor }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		scope:actor = {
			set_variable = { name = mbga_target_country value = root }
			remove_variable = mbga_active_state
			clear_variable_list = mbga_selectable
			clear_variable_list = mbga_taken
			set_variable = { name = mbga_taken_count value = 0 }
			set_variable = { name = mbga_loaded_count value = 0 }
			mbga_clear_candidates = yes
		}
		debug_log = "MBGA|api|target country set"
	}
}

# True once a target country has been chosen, so an interface can reveal its state list.
mbga_has_target_country = {
	scope = country
	is_shown = { has_variable = mbga_target_country }
	is_valid = { always = yes }
	effect = {}
}

# Two ways of gathering candidates. Normal keeps only provinces that touch the demander's
# own territory, which check_area answers per province because it takes a province scope.
# Godmode drops that filter and offers the whole state.
mbga_toggle_godmode = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = mbga_godmode }
			remove_variable = mbga_godmode
			debug_log = "MBGA|api|godmode off, bordering provinces only"
		}
		else = {
			set_variable = mbga_godmode
			debug_log = "MBGA|api|godmode on, every province of the state"
		}
		# takeable is decided as each candidate is made, so the list has to be rebuilt
		if = {
			limit = { has_variable = mbga_active_state }
			save_scope_as = actor
			var:mbga_active_state = { mbga_open_for_state_effect = yes }
		}
	}
}

mbga_godmode_on = {
	scope = country
	is_shown = { has_variable = mbga_godmode }
	is_valid = { always = yes }
	effect = {}
}
