# Dynamic probe: pick any state on the map, load its provinces, click them to
# transfer, click revert to put everything back.

# Fill the selectable list for whatever state the player has open.
mbga_open_for_state = {
	scope = state
	saved_scopes = { actor }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		debug_log = "MBGA|dyn|opening state region [ROOT.GetStateRegion.GetName]"
		scope:actor = {
			set_variable = { name = mbga_active_state value = ROOT }
			clear_variable_list = mbga_taken
			set_variable = { name = mbga_taken_count value = 0 }
		}
		mbga_fill_selectable_provinces = yes
		scope:actor = {
			set_variable = { name = mbga_loaded_count value = 0 }
			every_in_list = {
				variable = mbga_selectable
				scope:actor = { change_variable = { name = mbga_loaded_count add = 1 } }
			}
		}
		if = {
			limit = { scope:actor = { var:mbga_loaded_count > 0 } }
			debug_log = "MBGA|dyn|selectable list filled"
		}
		else = {
			debug_log = "MBGA|dyn|selectable list came back EMPTY"
		}
	}
}

# Click a province: hand it to the player right away and remember it.
mbga_take_province = {
	scope = province
	saved_scopes = { actor }
	is_shown = { always = yes }
	is_valid = {
		NOT = { scope:actor = { is_target_in_variable_list = { name = mbga_taken target = ROOT } } }
	}
	effect = {
		scope:actor = {
			add_to_variable_list = { name = mbga_taken target = ROOT }
			change_variable = { name = mbga_taken_count add = 1 }
			save_scope_as = mbga_taker
		}
		mbga_transfer_province_to_taker = yes
		debug_log = "MBGA|dyn|province taken"
	}
}

# Casing probe: the game's own data spells some province ids inconsistently.
mbga_probe_province_sgui = {
	scope = province
	saved_scopes = { actor }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		debug_log = "MBGA|06|province scripted gui fired, region [ROOT.GetStateRegion.GetName]"
	}
}

# Collapse toggle, so the panel can be tucked away when it is in the way.
mbga_toggle_panel = {
	scope = country
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = mbga_panel_collapsed }
			remove_variable = mbga_panel_collapsed
		}
		else = {
			set_variable = mbga_panel_collapsed
		}
	}
}
