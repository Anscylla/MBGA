# The clicked state, without touching a single vanilla file.
#
# target_type = state puts the player into the game's own state picker on the map
# (mm_diplomatic_play_state_interaction). Whatever they click arrives as scope:target_state
# in this war goal's own triggers, script values and effects. That scope reaches nowhere
# else: a diplomatic play exposes only .initiator and .target, on_wargoal_added carries no
# state, and the StatesPanel data context exists only inside the vanilla state panel.
#
# kind = custom means the game applies no territorial change of its own, leaving the
# per province transfer entirely to on_enforced.

mbga_demand_provinces = {
	icon = "gfx/interface/icons/war_goals/conquer_state.dds"

	kind = custom

	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		requires_interest
	}

	execution_priority = 68

	contestion_type = control_target_state

	target_type = state

	possible = {
	}

	valid = {
		scope:target_state = {
			owner = scope:target_country
		}
	}

	maneuvers = {
		value = 10
	}

	# Scaled by the target state's population, exactly as conquer_state does. Beyond
	# being reasonable, a number that visibly changes from one state to the next is the
	# proof that scope:target_state really resolves while the player is choosing.
	infamy = {
		value = 0

		add = {
			desc = "INFAMY_BASE_VALUE"
			value = 5
		}

		multiply = {
			desc = "INFAMY_STATE_POPULATION_FACTOR"
			value = scope:target_state.state_population
			divide = define:NDiplomacy|WAR_GOAL_INFAMY_POPULATION_SCALING_FACTOR
			multiply = define:NDiplomacy|WAR_GOAL_INFAMY_POPULATION_SCALING_MULTIPLIER
			add = 1
			max = define:NDiplomacy|WAR_GOAL_INFAMY_POPULATION_SCALING_MULTIPLIER_MAX_PER_STATE
		}

		min = {
			desc = "INFAMY_MINIMUM_VALUE"
			value = 5
		}
	}

	on_enforced = {
		scope:target_state = { save_scope_as = mbga_enforced_state }
		root = {
			set_variable = { name = mbga_last_demanded_state value = scope:mbga_enforced_state }
			save_scope_as = actor
		}
		scope:mbga_enforced_state = { mbga_fill_selectable_provinces = yes }

		if = {
			limit = { root = { any_in_list = { variable = mbga_selectable always = yes } } }
			debug_log = "MBGA|wg|target_state captured and its provinces were listed"
		}
		else = {
			debug_log = "MBGA|wg|target_state captured but the province list came back empty"
		}
	}

	ai = {
		is_significant_demand = no
	}
}
