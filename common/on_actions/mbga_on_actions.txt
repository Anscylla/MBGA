# Builds the province index at the start of a game.
#
# Nothing here settles a won state. Every way of winning one - a peace, a play conceded before
# the fighting, a demand granted outright - goes through the engine enforcing the war goal, so
# the goal's own on_enforced is the single place that draws a border.

on_game_started = {
	on_actions = {
		mbga_on_game_started
	}
}

mbga_on_game_started = {
	effect = {
		mbga_build_province_index = yes
	}
}
